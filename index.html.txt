        document.addEventListener('DOMContentLoaded', () => {
            
            // --- 1. Mobile Menu Toggle ---
            const mobileToggle = document.querySelector('.mobile-toggle');
            const navLinks = document.querySelector('.nav-links');

            mobileToggle.addEventListener('click', () => {
                navLinks.classList.toggle('active');
                // Change icon
                const icon = mobileToggle.querySelector('i');
                if (navLinks.classList.contains('active')) {
                    icon.classList.remove('fa-bars');
                    icon.classList.add('fa-times');
                } else {
                    icon.classList.remove('fa-times');
                    icon.classList.add('fa-bars');
                }
            });

            // Close menu when clicking a link
            document.querySelectorAll('.nav-links a').forEach(link => {
                link.addEventListener('click', () => {
                    navLinks.classList.remove('active');
                    mobileToggle.querySelector('i').classList.remove('fa-times');
                    mobileToggle.querySelector('i').classList.add('fa-bars');
                });
            });

            // --- 2. Scroll Reveal Animation ---
            const revealElements = document.querySelectorAll('.reveal');

            const revealOnScroll = () => {
                const windowHeight = window.innerHeight;
                const elementVisible = 150;

                revealElements.forEach((reveal) => {
                    const elementTop = reveal.getBoundingClientRect().top;
                    if (elementTop < windowHeight - elementVisible) {
                        reveal.classList.add('active');
                    }
                });
            };

            window.addEventListener('scroll', revealOnScroll);
            // Trigger once on load
            revealOnScroll();

            // --- 3. AI Demo Logic (Simulation) ---
            const aiForm = document.getElementById('aiForm');
            const aiResults = document.getElementById('ai-results');
            const loadingSpinner = document.getElementById('loading');

            aiForm.addEventListener('submit', (e) => {
                e.preventDefault();

                // Get Values
                const stream = document.getElementById('stream').value;
                const results = document.getElementById('results').value;
                const interest = document.getElementById('interest').value.toLowerCase();

                // Reset UI
                aiResults.style.display = 'none';
                aiResults.innerHTML = '';
                loadingSpinner.style.display = 'block';

                // Simulate Processing Delay (1.5 seconds)
                setTimeout(() => {
                    loadingSpinner.style.display = 'none';
                    generateRecommendations(stream, results, interest);
                }, 1500);
            });

            function generateRecommendations(stream, results, interest) {
                let recommendationsHTML = '';

                // Simple logic to mimic AI analysis based on inputs
                
                // 1. University Recommendation
                let uniText = "";
                if (results === '3A' || results === '2B1C') {
                    if(stream === 'bio') uniText = "High chance for Medicine (Colombo/Peradeniya) or Dental Surgery.";
                    else if(stream === 'maths') uniText = "Eligible for Engineering (Moratuwa) or Computer Science (UCSC/Peradeniya).";
                    else if(stream === 'commerce') uniText = "Eligible for Management (Jayewardanapura) or Finance (Colombo).";
                    else if(stream === 'arts') uniText = "Eligible for Arts (Peradeniya) or Law (Colombo).";
                    else uniText = "Strong chance for Engineering Technology or ICT based faculties.";
                } else if (results === '1S') {
                    uniText = "Consider applying for external degrees or private institutes (SLIIT, APIIT).";
                } else {
                    uniText = "Focus on Foundation courses or NVQ Level 5/6 vocational diplomas.";
                }

                // 2. Interest/Career Match
                let careerText = "";
                if (interest.includes('it') || interest.includes('software') || interest.includes('computer')) {
                    careerText = "Software Engineering, Data Science, or Cyber Security. High demand in SL.";
                } else if (interest.includes('nurse') || interest.includes('doc') || interest.includes('medical')) {
                    careerText = "Nursing (Private/Kandy), Pharmacy, or Lab Technology pathways.";
                } else if (interest.includes('teach') || interest.includes('edu')) {
                    careerText = "National College of Education (NCoE) or Arts Dip. in Education.";
                } else {
                    careerText = `Careers related to ${interest}. We suggest checking the TVEC database for specific vocational courses.`;
                }

                // 3. Alternative Pathway (Vocational/Abroad)
                let altText = "Scholarships available in India (low cost) for Engineering/MBBS.";
                if (stream === 'tech') altText = "City & Guilds or NVQ Level 6 in Mechatronics is a great alternative.";
                if (stream === 'arts') altText = "Consider HND in English/Marketing at SLIATE or private institutes.";

                // Build Cards
                recommendationsHTML += `
                    <div class="result-card">
                        <div class="result-title"><span>University Pathway</span> <i class="fas fa-graduation-cap"></i></div>
                        <p>${uniText}</p>
                    </div>
                    <div class="result-card">
                        <div class="result-title"><span>Career Match</span> <i class="fas fa-briefcase"></i></div>
                        <p>${careerText}</p>
                    </div>
                    <div class="result-card">
                        <div class="result-title"><span>Alternative Option</span> <i class="fas fa-route"></i></div>
                        <p>${altText}</p>
                    </div>
                `;

                aiResults.innerHTML = recommendationsHTML;
                aiResults.style.display = 'block';
            }
        });
